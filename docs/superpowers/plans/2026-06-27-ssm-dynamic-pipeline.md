# SSM 动态数据管线改造 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将 SSM 图层从静态 PNG 瓦片 + JSON 统计改造为 COG + TiTiler 动态瓦片 + rasterio 实时查询

**Architecture:** TiTiler TilerFactory 提供动态瓦片端点，rasterio 直接读取 COG 文件进行点查询和面积统计，前端 Leaflet 瓦片 URL 指向新的 TiTiler 端点。

**Tech Stack:** Python 3.13, FastAPI, titiler.core, rasterio, rio-cogeo, React 19, Leaflet, TypeScript

## Global Constraints

- 仅改造 SSM 图层，其他 4 个图层（NDVI、降水、土壤湿度、LST）保持原逻辑不变
- 旧的 `/data/tiles/` 静态瓦片路由保持可用
- SSM 需同时支持 8 天合成（46 期/年）和月均值两种时间分辨率
- 时间格式：8 天合成用 `YYYY_NN`（如 `2010_05`），月均值用 `YYYY-MM`（如 `2010-02`）
- 前端 colormap 使用 `rdylgn`（与 SSM 图例颜色匹配）

### Task 2: COG 批量转换脚本

**Files:**

**Interfaces:**
- Produces: 脚本把 `F:\全国灌溉用水反演\数据2010-2013\SSM预测结果\*.tif` 转为 `data/rasters/ssm/*_cog.tif`

- [ ] **Step 1: 编写转换脚本**

```python
"""Batch convert SSM GeoTIFF files to Cloud-Optimized GeoTIFF (COG) format."""

import subprocess
import sys
from pathlib import Path

SRC_DIR = Path(r"F:\全国灌溉用水反演\数据2010-2013\SSM预测结果")
DST_DIR = Path(__file__).resolve().parent.parent / "data" / "rasters" / "ssm"


def convert_to_cog(src: Path, dst: Path) -> bool:
    """Convert a single GeoTIFF to COG using rio cogeo create."""
    dst.parent.mkdir(parents=True, exist_ok=True)
    # Strip .aux.xml if present (it's created by GDAL but not needed for COG)
    cmd = [
        "rio", "cogeo", "create",
        str(src),
        str(dst),
        "--cog-profile", "deflate",
        "--overwrite",
    ]
    result = subprocess.run(cmd, capture_output=True, text=True)
    if result.returncode != 0:
        print(f"  FAIL {src.name}: {result.stderr.strip()}")
        return False
    print(f"  OK   {src.name} -> {dst.name}")
    return True


def main():
    tif_files = sorted(SRC_DIR.glob("*.tif"))
    if not tif_files:
        print(f"No .tif files found in {SRC_DIR}")
        sys.exit(1)

    print(f"Found {len(tif_files)} TIF files. Converting to COG...")
    ok = 0
    fail = 0
    for tif in tif_files:
        dst = DST_DIR / f"{tif.stem}_cog.tif"
        if convert_to_cog(tif, dst):
            ok += 1
        else:
            fail += 1

    print(f"\nDone: {ok} succeeded, {fail} failed")


if __name__ == "__main__":
    main()
```

```python
"""Dynamic tile serving via TiTiler — Cloud-Optimized GeoTIFF (COG) rendering."""

from pathlib import Path

from fastapi import APIRouter, HTTPException, status
from fastapi.responses import FileResponse
from titiler.core.factory import TilerFactory

router = APIRouter(tags=["tiles"])

# ===== TiTiler dynamic COG tile engine =====
# All endpoints under this prefix are auto-generated by TilerFactory:
#   GET  /cog/tiles/{z}/{x}/{y}.png
#   GET  /cog/info
#   GET  /cog/point/{lon},{lat}
#   etc.

cog = TilerFactory(
    router_prefix="/cog",
    # Allow URL parameter to specify COG file path (relative to project root)
)
cog_tiler = cog.router

# ===== Legacy static tile serving (for pre-generated PNG tiles) =====

# Project root is two levels up from routers/tiles.py → backend/routers/ → project/
PROJECT_ROOT = Path(__file__).resolve().parent.parent.parent
TILES_ROOT = PROJECT_ROOT / "data" / "tiles"


@router.get("/tiles/{layer}/{time}/{z}/{x}/{y}.png")
def serve_static_tile(layer: str, time: str, z: int, x: int, y: int):
    """Serve a pre-generated raster tile image (legacy path for non-SSM layers)."""
    tile_path = TILES_ROOT / layer / time / str(z) / str(x) / f"{y}.png"
    if not tile_path.is_file():
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail=f"Tile not found: {tile_path}",
        )
    return FileResponse(tile_path, media_type="image/png")
```

```python

# ---- SSM 图层使用 rasterio 读取 COG 文件 ----
if layerId == "ssm":
    from pathlib import Path
    import rasterio
    from pyproj import Transformer

    PROJECT_ROOT = Path(__file__).resolve().parent.parent.parent
    cog_path = PROJECT_ROOT / "data" / "rasters" / "ssm" / f"{time}_cog.tif"

    if not cog_path.is_file():
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail=f"COG file not found: {cog_path}",
        )

    try:
        with rasterio.open(cog_path) as src:
            transformer = Transformer.from_crs(
                "EPSG:4326", src.crs, always_xy=True
            )
            x, y = transformer.transform(lng, lat)
            row, col = src.index(x, y)
            # 检查行列号是否在数据范围内
            if row < 0 or row >= src.height or col < 0 or col >= src.width:
                raise HTTPException(
                    status_code=status.HTTP_404_NOT_FOUND,
                    detail="Point is outside the raster extent",
                )
            val = src.read(1, window=((row, row + 1), (col, col + 1)))
            value = float(val[0, 0])
            # NaN 或 nodata 视为无数据
            import math
            if math.isnan(value) or (
                src.nodata is not None and value == src.nodata
            ):
                raise HTTPException(
                    status_code=status.HTTP_404_NOT_FOUND,
                    detail="No valid data at this point",
                )
    except rasterio.errors.RasterioIOError as e:
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail=f"Failed to read raster: {e}",
        )

    return {
        "layerId": layerId,
        "time": time,
        "lng": lng,
        "lat": lat,
        "value": value,
        "unit": layer["unit"],
    }

# ---- 原有的 JSON stats 查询逻辑（非 SSM 图层） ----
region = _find_region_for_point(lng, lat)
# ... 后面的代码保持不变
```

### Task 5: 后端 — rasterio 实时面积查询

**Files:**
- Modify: `backend/routers/query.py`

**Interfaces:**
- Consumes: `POST /api/query/area {layerId, time, geometry}`，COG 文件
- Produces: `{mean, max, min, count}` — 矩形内有效像元的统计值

- [ ] **Step 1: 添加 rasterio 面积查询逻辑**

在 `area_query` 函数中，`layer` 验证通过后、bbox 计算之前，插入 SSM 分支：

```python
# ---- SSM 图层使用 rasterio 读取 COG 文件进行面积统计 ----
if body.layerId == "ssm":
    from pathlib import Path
    import rasterio
    from rasterio.warp import transform_bounds
    import numpy as np
    import math

    PROJECT_ROOT = Path(__file__).resolve().parent.parent.parent
    cog_path = PROJECT_ROOT / "data" / "rasters" / "ssm" / f"{body.time}_cog.tif"

    if not cog_path.is_file():
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail=f"COG file not found: {cog_path}",
        )

    # 解析 GeoJSON geometry 的 bounding box
    if body.geometry.type != "Polygon":
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            detail="Only Polygon geometry type is supported for area queries",
        )
    try:
        ring = body.geometry.coordinates[0]
        lngs = [pt[0] for pt in ring]
        lats = [pt[1] for pt in ring]
        west, south, east, north = min(lngs), min(lats), max(lngs), max(lats)
    except (IndexError, TypeError):
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            detail="Invalid geometry coordinates",
        )

    try:
        with rasterio.open(cog_path) as src:
            # 将 WGS84 bbox 转为栅格行列号
            row_min, col_min = src.index(west, north)
            row_max, col_max = src.index(east, south)
            # 确保行列号在有效范围内
            row_min, row_max = max(0, row_min), min(src.height, row_max + 1)
            col_min, col_max = max(0, col_min), min(src.width, col_max + 1)
            if row_min >= row_max or col_min >= col_max:
                raise HTTPException(
                    status_code=status.HTTP_404_NOT_FOUND,
                    detail="Area is outside the raster extent",
                )
            data = src.read(
                1, window=((row_min, row_max), (col_min, col_max))
            )
            # 过滤 NaN 和无效值
            valid = data[~np.isnan(data)]
            if src.nodata is not None:
                valid = valid[valid != src.nodata]
            if valid.size == 0:
                raise HTTPException(
                    status_code=status.HTTP_404_NOT_FOUND,
                    detail="No valid data in the specified area",
                )
            return {
                "mean": float(valid.mean()),
                "max": float(valid.max()),
                "min": float(valid.min()),
                "count": int(valid.size),
            }
    except rasterio.errors.RasterioIOError as e:
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail=f"Failed to read raster: {e}",
        )
```
